# MdView Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Introduce `View` enum (Editor | Markdown) to make markdown preview state per-tab instead of global on UiShell.

**Architecture:** `DocumentView` is the base class; `MdView` = `DocumentView` + `MdPreviewState` is the derived class. `View` enum holds either variant. Workspace stores `Vec<View>`. Render/scroll branches match on `View` instead of checking a global bool.

**Tech Stack:** Rust, winit + wgpu, edit_plus_markdown crate

**Prerequisites:** `crates/markdown/` (edit_plus_markdown) must exist. If not present, bring it from the `codex/markdown-preview-rendering` branch first.

// crates/app/src/view.rs

use edit_plus_markdown::layout::LaidOutDoc;
use edit_plus_markdown::style::MarkdownStyle;
use ui::core::paint::DrawList;
use ui::Theme;
use crate::document_view::DocumentView;
use std::path::PathBuf;

/// Preview-only state: caching, layout, scroll.
pub(crate) struct MdPreviewState {
    pub scroll_y: f32,
    pub content_height: f32,
    source: String,
    cached_layout: Option<LaidOutDoc>,
    cached_style_hash: u64,
    cached_source_hash: u64,
    cached_generation: u32,
    dirty: bool,
}

impl MdPreviewState {
    pub fn new() -> Self {
        Self {
            scroll_y: 0.0,
            content_height: 0.0,
            source: String::new(),
            cached_layout: None,
            cached_style_hash: 0,
            cached_source_hash: 0,
            cached_generation: 0,
            dirty: true,
        }
    }

    /// O(1) check: has the gap buffer generation changed?
    pub fn needs_source_update(&self, generation: u32) -> bool {
        generation != self.cached_generation
    }

    /// Update source text. Only marks dirty if content actually changed (hash check).
    pub fn set_source(&mut self, text: String, generation: u32) {
        let hash = fxhash(&text);
        if hash != self.cached_source_hash {
            self.source = text;
            self.cached_source_hash = hash;
            self.dirty = true;
        }
        self.cached_generation = generation;
    }

    /// Scroll by delta pixels. Returns true if scroll position changed.
    pub fn scroll(&mut self, delta: f32, viewport_h: f32) -> bool {
        let max_scroll = (self.content_height - viewport_h).max(0.0);
        let old = self.scroll_y;
        self.scroll_y = (self.scroll_y + delta).clamp(0.0, max_scroll);
        (self.scroll_y - old).abs() > 0.5
    }

    /// Render, cached. Re-parses only when source or style changed.
    pub fn render(
        &mut self,
        theme: &Theme,
        viewport_w: f32,
        viewport_h: f32,
        offset_x: f32,
        offset_y: f32,
    ) -> DrawList {
        let style = MarkdownStyle::from_theme(
            theme,
            ui::settings::Settings::with(|s| s.font_size),
            ui::settings::Settings::with(|s| s.line_height),
        );
        let style_hash = style_hash_quick(&style);

        if self.dirty || self.cached_layout.is_none() || style_hash != self.cached_style_hash {
            let parsed = edit_plus_markdown::parser::parse_markdown(&self.source);
            let doc = edit_plus_markdown::builder::MarkdownDoc::build(&parsed, &style);
            let laid_out = edit_plus_markdown::layout::layout_doc(&doc, &style, viewport_w);
            self.content_height = laid_out.total_height;
            self.cached_layout = Some(laid_out);
            self.cached_style_hash = style_hash;
            self.dirty = false;
        }

        let laid_out = self.cached_layout.as_ref().unwrap();
        let mut dl = DrawList::new();
        edit_plus_markdown::render::render_doc_with_offset(
            laid_out, &style, &mut dl, self.scroll_y, viewport_h, offset_x, offset_y,
        );
        dl
    }
}

/// DocumentView + preview capability.
pub(crate) struct MdView {
    pub doc: DocumentView,
    pub preview: MdPreviewState,
}

impl MdView {
    pub fn new(doc: DocumentView) -> Self {
        Self { doc, preview: MdPreviewState::new() }
    }

    pub fn into_doc(self) -> DocumentView {
        self.doc
    }
}

/// View replaces DocumentView as the unit stored in Workspace.
pub(crate) enum View {
    Editor(DocumentView),
    Markdown(MdView),
}

impl View {
    pub fn doc(&self) -> &DocumentView {
        match self {
            View::Editor(dv) => dv,
            View::Markdown(mv) => &mv.doc,
        }
    }

    pub fn doc_mut(&mut self) -> &mut DocumentView {
        match self {
            View::Editor(dv) => dv,
            View::Markdown(mv) => &mut mv.doc,
        }
    }

    pub fn file_path(&self) -> Option<&PathBuf> {
        self.doc().file_path.as_ref()
    }

    pub fn is_dirty(&self) -> bool {
        self.doc().dirty
    }

    pub fn is_markdown(&self) -> bool {
        matches!(self, View::Markdown(_))
    }

    pub fn into_editor(self) -> Self {
        match self {
            View::Markdown(mv) => View::Editor(mv.into_doc()),
            other => other,
        }
    }

    pub fn into_markdown(self) -> Self {
        match self {
            View::Editor(dv) => View::Markdown(MdView::new(dv)),
            other => other,
        }
    }
}

// ── Internal: fast hashing ──

fn fxhash(s: &str) -> u64 {
    let mut h: u64 = 0x517cc1b727220a95;
    for byte in s.bytes() {
        h = h.rotate_left(5) ^ (byte as u64);
    }
    h
}

fn style_hash_quick(style: &MarkdownStyle) -> u64 {
    let mut h: u64 = 0x517cc1b727220a95;
    for &fs in &style.heading_font_sizes {
        h = h.rotate_left(5) ^ (fs.to_bits() as u64);
    }
    h = h.rotate_left(5) ^ (style.body_font_size.to_bits() as u64);
    h = h.rotate_left(5) ^ (style.code_font_size.to_bits() as u64);
    h = h.rotate_left(5) ^ (style.line_height.to_bits() as u64);
    h = h.rotate_left(5) ^ (style.paragraph_spacing.to_bits() as u64);
    h = h.rotate_left(5) ^ (style.list_indent.to_bits() as u64);
    for &c in &[style.text_color, style.code_color, style.code_bg, style.link_color,
                style.heading_color, style.blockquote_color, style.blockquote_border] {
        for &v in &c {
            h = h.rotate_left(5) ^ (v.to_bits() as u64);
        }
    }
    h
}

// ── Markdown preview mode ──
if let Some(View::Markdown(mv)) = self.workspace.views.get_mut(self.workspace.active_index) {
    // Update source from document (only if buffer generation changed)
    let buf_gen = mv.doc.tb().gap_buffer().generation();
    if mv.preview.needs_source_update(buf_gen) {
        let gb = mv.doc.tb().gap_buffer();
        let c1 = gb.read_forward(0);
        let c2 = gb.read_forward(c1.len());
        let mut text = String::with_capacity(c1.len() + c2.len());
        text.push_str(std::str::from_utf8(c1).unwrap_or(""));
        text.push_str(std::str::from_utf8(c2).unwrap_or(""));
        mv.preview.set_source(text, buf_gen);
    }

    let editor_r = self.ui_shell.editor_rect();
    let scrollbar_r = Settings::with(|s| s.scrollbar_reserve());
    let content_right = editor_r.x + editor_r.w - scrollbar_r;
    let viewport_w = (content_right - gutter_left_margin).max(100.0);
    let viewport_h = editor_r.h;
    let offset_x = gutter_left_margin;
    let offset_y = content_top;
    let dl = mv.preview.render(&self.current_theme, viewport_w, viewport_h, offset_x, offset_y);

    // Update scrollbar for preview content
    let line_height = Settings::with(|s| s.line_height);
    let total_rows = (mv.preview.content_height / line_height).ceil() as usize;
    let scroll_y = mv.preview.scroll_y as f64;
    self.ui_shell.set_scrollbar_input(viewport_h as f64, total_rows, scroll_y);

    // Convert DrawList to vertices via paint_backend
    let screen = ui::core::Screen::new(screen_w, screen_h);
    if let (Some(text), Some(gpu)) = (self.text.as_mut(), self.gpu.as_ref()) {
        let preview_verts = crate::paint_backend::drain(dl, screen, Some(text), Some(gpu));
        vertices.extend(preview_verts);
    }

    // Skip normal editor rendering for this frame
    self.needs_redraw = true;
    return;
}
