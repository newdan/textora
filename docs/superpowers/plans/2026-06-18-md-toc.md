```rust
/// Collect headings from the laid-out document. Called after layout completes.
fn collect_headings(&mut self) {
    self.headings.clear();
    let mut y: f32 = 0.0;
    for (bi, block) in self.lazy.laid_blocks.iter().enumerate() {
        let doc_idx = self.lazy.laid_to_doc[bi];
        let kind = match self.lazy.doc.blocks.get(doc_idx) {
            Some(b) => &b.kind,
            None => {
                y += block.height();
                continue;
            }
        };
        if let edit_plus_markdown::builder::BlockKind::Heading { level } = kind {
            let text = match block {
                LaidOutBlock { kind: LaidOutBlockKind::Text { lines }, .. } => {
                    lines.first().map(|l| l.text().to_string()).unwrap_or_default()
                }
                _ => String::new(),
            };
            self.headings.push(HeadingEntry {
                text,
                level: *level,
                block_idx: bi,
                y_offset: y,
            });
        }
        y += block.height();
    }
}
```

Call `self.collect_headings()` in `render()` after layout completes (after `self.cache_dirty = false`).

- [ ] **Step 3: Add `headings()`, `current_heading_index()`, and `scroll_to_heading()` accessors**

```rust
pub fn headings(&self) -> &[HeadingEntry] {
    &self.headings
}

/// Return the index into headings() for the heading closest to scroll_y.
pub fn current_heading_index(&self, scroll_y: f32) -> Option<usize> {
    if self.headings.is_empty() {
        return None;
    }
    match self.headings.binary_search_by(|h| {
        h.y_offset.partial_cmp(&scroll_y).unwrap_or(std::cmp::Ordering::Equal)
    }) {
        Ok(i) => Some(i),
        Err(0) => Some(0),
        Err(i) => Some(i - 1),
    }
}

/// Set scroll_y to bring a heading to the top of the viewport.
pub fn scroll_to_heading(&mut self, index: usize) {
    if let Some(h) = self.headings.get(index) {
        self.scroll_y = h.y_offset;
        self.clamp_scroll();
    }
}
```

```rust
//! TocWidget — markdown preview table of contents panel.
//!
//! Renders a hierarchical heading list as a Dock Left child.
//! Supports click-to-jump, scroll tracking (current heading highlight),
//! and hover feedback.

use winit::window::CursorIcon;
use crate::core::{Widget, Rect, LayoutCtx, PaintCtx, EventCtx, Event, MouseButton};
use crate::core::widget::WidgetAction;
use crate::settings::Settings;
use std::any::Any;

/// Indent per heading level in physical pixels.
const INDENT_PER_LEVEL: f32 = 16.0;
/// Row height per heading entry in logical pixels.
const ROW_HEIGHT: f32 = 24.0;
/// Font size for heading entries in logical pixels.
const FONT_SIZE: f32 = 12.0;
/// Horizontal padding inside the panel in logical pixels.
const H_PAD: f32 = 8.0;
/// Active indicator bar width in logical pixels.
const ACTIVE_BAR_WIDTH: f32 = 2.0;

/// Per-frame input for the TOC widget.
#[derive(Clone, Debug)]
pub struct TocInput {
    /// Heading entries extracted from the markdown document.
    pub headings: Vec<TocHeadingEntry>,
    /// Index of the currently active heading (closest to viewport top).
    pub active_index: Option<usize>,
}

/// A single heading entry in the TOC (UI crate version).
#[derive(Clone, Debug)]
pub struct TocHeadingEntry {
    pub text: String,
    pub level: u8,
}

/// TocWidget: renders a scrollable hierarchical heading list.
pub struct TocWidget {
    rect: Rect,
    input: Option<TocInput>,
    /// Vertical scroll offset of the TOC panel itself.
    scroll_y: f32,
    /// Hovered heading index (for hover highlight).
    hovered_index: Option<usize>,
    /// Precomputed row rects for hit testing.
    row_rects: Vec<Rect>,
}

impl TocWidget {
    pub fn new() -> Self {
        Self {
            rect: Rect::ZERO,
            input: None,
            scroll_y: 0.0,
            hovered_index: None,
            row_rects: Vec::new(),
        }
    }

    pub fn set_input(&mut self, input: TocInput) {
        self.input = Some(input);
    }
}

impl Widget for TocWidget {
    fn set_rect(&mut self, rect: Rect, ctx: &mut LayoutCtx) {
        self.rect = Rect::new(0.0, 0.0, rect.w, rect.h);
        self.row_rects.clear();
        let dpi = ctx.dpi;
        let row_h = ROW_HEIGHT * dpi;
        let w = rect.w;
        let max_depth = Settings::with(|s| s.toc_max_depth);
        let input = match &self.input {
            Some(inp) => inp,
            None => return,
        };
        let mut y: f32 = 4.0 * dpi - self.scroll_y;
        for h in input.headings.iter() {
            if h.level > max_depth {
                continue;
            }
            let indent = (h.level as f32 - 1.0) * INDENT_PER_LEVEL * dpi;
            let rx = H_PAD * dpi + indent;
            if y + row_h > 0.0 && y < rect.h {
                self.row_rects.push(Rect::new(rx, y, w - rx - 4.0 * dpi, row_h));
            } else {
                self.row_rects.push(Rect::ZERO);
            }
            y += row_h;
        }
    }

    fn paint(&self, ctx: &mut PaintCtx) {
        if self.rect.w <= 0.0 || self.rect.h <= 0.0 {
            return;
        }
        let Some(ref input) = self.input else { return };

        let dpi = ctx.dpi;
        let row_h = ROW_HEIGHT * dpi;
        let font_size = FONT_SIZE * dpi;
        let max_depth = Settings::with(|s| s.toc_max_depth);

        // Background (use sidebar bg for consistency)
        ctx.list.fill(self.rect, ctx.theme.sidebar_bg);

        if input.headings.iter().all(|h| h.level > max_depth) {
            let mut color = ctx.theme.sidebar_item_fg;
            color[3] *= 0.4;
            let empty_text = "No headings";
            if let Some(ref mut shaper) = ctx.shaper {
                ctx.list.text_shaped(
                    self.rect.x + 20.0 * dpi,
                    self.rect.y + self.rect.h * 0.4,
                    font_size, color, empty_text, shaper,
                );
            }
            return;
        }

        let mut y: f32 = self.rect.y + 4.0 * dpi - self.scroll_y;

        for (i, h) in input.headings.iter().enumerate() {
            if h.level > max_depth {
                continue;
            }
            if y + row_h < self.rect.y {
                y += row_h;
                continue;
            }
            if y > self.rect.y + self.rect.h {
                break;
            }

            let indent = (h.level as f32 - 1.0) * INDENT_PER_LEVEL * dpi;
            let x = self.rect.x + H_PAD * dpi + indent;

            let is_active = input.active_index == Some(i);
            let is_hovered = self.hovered_index == Some(i);

            // Active indicator bar
            if is_active {
                let bar_x = self.rect.x + 2.0 * dpi;
                let bar_h = row_h * 0.6;
                let bar_y = y + (row_h - bar_h) * 0.5;
                ctx.list.fill_rounded(
                    Rect::new(bar_x, bar_y, ACTIVE_BAR_WIDTH * dpi, bar_h),
                    ctx.theme.sidebar_accent,
                    1.0 * dpi,
                );
            }

            // Hover background
            if is_hovered && !is_active {
                let mut hover_bg = ctx.theme.sidebar_item_fg;
                hover_bg[3] *= 0.08;
                ctx.list.fill(Rect::new(self.rect.x, y, self.rect.w, row_h), hover_bg);
            }

            // Heading text color
            let color = if h.level <= 2 {
                ctx.theme.sidebar_accent
            } else {
                ctx.theme.sidebar_item_fg
            };

            if let Some(ref mut shaper) = ctx.shaper {
                let weight = if is_active {
                    shaping::Weight::SEMIBOLD
                } else {
                    shaping::Weight::NORMAL
                };
                ctx.list.text_shaped_with_font(
                    x, y + row_h * 0.7,
                    font_size, color, &h.text,
                    None, weight, shaping::Style::Normal,
                    false, shaper,
                );
            }

            y += row_h;
        }
    }

    fn hit(&self, px: f32, py: f32) -> bool {
        self.rect.contains(px, py)
    }

    fn on_event(&mut self, ev: &Event, ctx: &mut EventCtx) -> Option<WidgetAction> {
        match ev {
            Event::MouseMove { px, py } => {
                if !self.rect.contains(*px, *py) {
                    self.hovered_index = None;
                    return None;
                }
                let local_y = py - self.rect.y + self.scroll_y - 4.0 * ctx.dpi;
                let row_h = ROW_HEIGHT * ctx.dpi;
                let row = (local_y / row_h).floor() as i32;
                if row >= 0 && (row as usize) < self.row_rects.len()
                    && self.row_rects[row as usize] != Rect::ZERO
                {
                    // Map visible row back to original heading index
                    let max_depth = Settings::with(|s| s.toc_max_depth);
                    let input = self.input.as_ref().unwrap();
                    let mut visible_count: i32 = -1;
                    let mut heading_idx = None;
                    for (hi, h) in input.headings.iter().enumerate() {
                        if h.level > max_depth {
                            continue;
                        }
                        visible_count += 1;
                        if visible_count == row {
                            heading_idx = Some(hi);
                            break;
                        }
                    }
                    self.hovered_index = heading_idx;
                } else {
                    self.hovered_index = None;
                }
                ctx.cursor_hint = Some(CursorIcon::Pointer);
                Some(WidgetAction::Consumed)
            }
            Event::MouseDown { px, py, button } => {
                if *button != MouseButton::Left || !self.rect.contains(*px, *py) {
                    return Some(WidgetAction::Consumed);
                }
                let local_y = py - self.rect.y + self.scroll_y - 4.0 * ctx.dpi;
                let row_h = ROW_HEIGHT * ctx.dpi;
                let row = (local_y / row_h).floor() as usize;
                if let Some(ref input) = self.input {
                    let max_depth = Settings::with(|s| s.toc_max_depth);
                    let mut visible_count = 0;
                    for (hi, h) in input.headings.iter().enumerate() {
                        if h.level > max_depth {
                            continue;
                        }
                        if visible_count == row {
                            return Some(WidgetAction::Toc(TocAction::JumpToHeading(hi)));
                        }
                        visible_count += 1;
                    }
                }
                Some(WidgetAction::Consumed)
            }
            Event::Wheel { dy, .. } => {
                let dpi = ctx.dpi;
                let max_scroll = (self.row_rects.len() as f32 * ROW_HEIGHT * dpi
                    + 8.0 * dpi - self.rect.h).max(0.0);
                self.scroll_y = (self.scroll_y + dy).clamp(0.0, max_scroll);
                Some(WidgetAction::Consumed)
            }
            Event::MouseUp { .. } => {
                Some(WidgetAction::Consumed)
            }
            _ => None,
        }
    }

    fn as_any(&self) -> &dyn Any {
        self
    }

    fn as_any_mut(&mut self) -> &mut dyn Any {
        self
    }
}

/// Actions emitted by TOC widget.
#[derive(Debug, Clone, PartialEq)]
pub enum TocAction {
    /// User clicked a heading to jump to it.
    JumpToHeading(usize),
}
```
